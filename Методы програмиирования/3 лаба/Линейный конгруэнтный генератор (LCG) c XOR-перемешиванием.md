```cpp
class ModifiedLCGPermuted : public Generator {
    uint64_t state;

public:
    ModifiedLCGPermuted(uint64_t seed)
        : state(seed ? seed : 1) {}

    uint32_t nextU32() override {
        // state = a * state + c mod 2^64
        state = state * 6364136223846793005ULL + 1442695040888963407ULL;

        uint64_t z = state;

        // Модификация: отдаём не само состояние LCG,
        // а дополнительно перемешанное значение.
        // Это улучшает поведение битов, особенно младших.
        z ^= z >> 30;
        z *= 0xbf58476d1ce4e5b9ULL;

        z ^= z >> 27;
        z *= 0x94d049bb133111ebULL;

        z ^= z >> 31;

        // Возвращаем старшие 32 бита перемешанного 64-битного значения.
        return uint32_t(z >> 32);
    }

    string name() const override {
        return "ModifiedLCGPermuted";
    }
};
```
Второй генератор основан на линейном конгруэнтном методе. Его внутреннее состояние обновляется по формуле `Xₙ₊₁ = aXₙ + c mod 2^64`. Модуль явно не записан, потому что для `uint64_t` переполнение автоматически даёт арифметику по модулю 2^64. Чтобы уменьшить недостатки классического LCG, особенно плохое качество младших битов, я не возвращаю состояние напрямую, а пропускаю его через дополнительную функцию перемешивания из XOR-сдвигов и умножений на большие 64-битные константы. После этого возвращаются старшие 32 бита результата.
