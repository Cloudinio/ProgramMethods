```cpp
class XorShift32Salt : public Generator {
private:
    uint32_t x;
    uint32_t w;

public:
    XorShift32Salt(uint32_t seed)
        : x(seed ? seed : 2463534242u),
          w(0x9E3779B9u) {}

    uint32_t nextU32() override {
        x ^= x << 13;
        x ^= x >> 17;
        x ^= x << 5;

        w += 0x9E3779B9u;

        return x + w;
    }

    string name() const override {
        return "XorShift32Salt";
    }
};
```
Генератор `XorShift32Salt` основан на `xorshift`. Он хранит 32-битное состояние `x`, которое на каждом шаге перемешивается с помощью `XOR` и битовых сдвигов. Чтобы усилить изменение выходной последовательности, добавлена дополнительная переменная `w`, которую можно рассматривать как соль. Она на каждом шаге увеличивается на фиксированную константу. На выход возвращается сумма `x + w`, поэтому результат зависит не только от xorshift-состояния, но и от дополнительной изменяющейся компоненты.
